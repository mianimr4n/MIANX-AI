   703→function TestRunnerTab({ title, endpoint, result, loading, onRunTest }: { title: string; endpoint: string; result: TestResult | null; loading: boolean; onRunTest: () => void }) {
   704→  return (
   705→    <div className="space-y-4">
   706→      <Card className="border-border/50">
   707→        <CardHeader className="pb-3"><CardTitle className="text-base font-semibold">{title}</CardTitle><CardDescription>POST {endpoint}</CardDescription></CardHeader>
   708→        <CardContent className="flex items-center gap-3">
   709→          <Button onClick={onRunTest} disabled={loading} size="sm">
   710→            {loading ? <><Loader2 className="w-4 h-4 mr-1.5 animate-spin" />Running...</> : <><Play className="w-4 h-4 mr-1.5" />Run Tests</>}
   711→          </Button>
   712→          {result && (
   713→            <div className="flex items-center gap-2 text-sm">
   714→              <Badge variant={result.summary.failed === 0 ? 'default' : 'destructive'}>
   715→                {result.summary.passed}/{result.summary.total} passed
   716→              </Badge>
   717→              {result.summary.failed > 0 && <Badge variant="destructive">{result.summary.failed} failed</Badge>}
   718→            </div>
   719→          )}
   720→        </CardContent>
   721→      </Card>
   722→      {result && (
   723→        <Card className="border-border/50">
   724→          <CardHeader className="pb-3"><CardTitle className="text-base font-semibold">Test Results</CardTitle></CardHeader>
   725→          <CardContent>
   726→            <div className="space-y-1.5">
   727→              {result.results.map((t, i) => (
   728→                <div key={i} className="flex items-center gap-2 text-sm py-1">
   729→                  {t.passed ? <CheckCircle2 className="w-4 h-4 text-emerald-500 shrink-0" /> : <XCircle className="w-4 h-4 text-red-500 shrink-0" />}
   730→                  <span className="flex-1 font-medium">{t.name}</span>
   731→                  <span className="text-xs text-muted-foreground max-w-xs truncate">{t.detail}</span>
   732→                </div>
   733→              ))}
   734→            </div>
   735→          </CardContent>
   736→        </Card>
   737→      )}
   738→    </div>
   739→  )
